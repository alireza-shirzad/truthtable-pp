\section{Deferred PIOPs}
\label{sec:deferred-piops}
This appendix collects the low-level PIOPs and relations used throughout this paper. These are standard or lightly adapted from prior work, and are deferred here so that the main text can focus on the string-filtering protocols.

\subsection{NoZero Check}
The \emph{NoZero Check} takes a column $\col = (\mathsf{a},\mathsf{d})$ and proves that every active entry of $\col$ is nonzero. We package this as the relation
\ifsp
%% S&P (two-column): tuple and condition stacked so the display fits one column.
\begin{displaymath}
\NPRelation_{\mathsf{NoZero}} = \left\{\,
  \begin{aligned}
    & \NPInstance = \bot,\\
    & \NPInstancey = (\Oracle{\hat{\col}}),\\
    & \NPWitness = (\col)\\[2pt]
    & \text{s.t.}\ \forall i,\, \mathsf{a}[i] = 1 \implies \mathsf{d}[i] \neq 0
  \end{aligned}
\,\right\}.
\end{displaymath}
\else
\begin{displaymath}
\NPRelation_{\mathsf{NoZero}} = \left\{\,
  \NPInstance = \bot,\;
  \NPInstancey = (\Oracle{\hat{\col}}),\;
  \NPWitness = (\col)
  \;\middle|\;
    \forall i,\, \mathsf{a}[i] = 1 \implies \mathsf{d}[i] \neq 0
\,\right\}.
\end{displaymath}
\fi

A field element is nonzero exactly when it has a multiplicative inverse. The prover sends an inverse column $\mathsf{inv}$ whose value at each active index is the inverse of the data there and which is $0$ at inactive indices. If $\mathsf{inv}$ is formed this way, then $\mathsf{d}\mult\mathsf{inv} = \mathsf{a}$ holds across the whole hypercube: at an active index both sides equal $1$, and at an inactive index both sides equal $0$ (since $\mathsf{inv}$ is $0$ there). A single \cref{piop:zerocheck} on $\mathsf{d}\mult\mathsf{inv} - \mathsf{a}$ certifies this, and forces every active entry to be invertible, hence nonzero.
\begin{DefinePIOP}{NoZero Check}{nozero-check}
  \emph{Inputs:} $\Oracle{\hat{\col}} = (\Oracle{\MLE{\mathsf{a}}}, \Oracle{\MLE{\mathsf{d}}})$.
  \begin{enumerate}[nosep]
    \item $\PIOPProver$ constructs and sends the inverse oracle $\Oracle{\MLE{\mathsf{inv}}}$, with $\mathsf{inv}[i] = \mathsf{d}[i]^{-1}$ at active indices and $\mathsf{inv}[i] = 0$ elsewhere.
    \item $\PIOPProver$ and $\PIOPVerifier$ invoke \cref{piop:zerocheck} on the column $\mathsf{d}\mult\mathsf{inv} - \mathsf{a}$.
  \end{enumerate}
\end{DefinePIOP}

\subsection{Booleanity Check}
The \emph{Booleanity Check} takes a column $\col = (\mathsf{a},\mathsf{d})$ and proves that every active entry of $\col$ is $0$ or $1$. We package this as the relation
\ifsp
%% S&P (two-column): tuple and condition stacked so the display fits one column.
\begin{displaymath}
\NPRelation_{\mathsf{Bool}} = \left\{\,
  \begin{aligned}
    & \NPInstance = \bot,\\
    & \NPInstancey = (\Oracle{\hat{\col}}),\\
    & \NPWitness = (\col)\\[2pt]
    & \text{s.t.}\ \forall i,\, \mathsf{a}[i] = 1 \implies \mathsf{d}[i]\in\{0,1\}
  \end{aligned}
\,\right\}.
\end{displaymath}
\else
\begin{displaymath}
\NPRelation_{\mathsf{Bool}} = \left\{\,
  \NPInstance = \bot,\;
  \NPInstancey = (\Oracle{\hat{\col}}),\;
  \NPWitness = (\col)
  \;\middle|\;
    \forall i,\, \mathsf{a}[i] = 1 \implies \mathsf{d}[i]\in\{0,1\}
\,\right\}.
\end{displaymath}
\fi
Since $\mathsf{d}[i]\in\{0,1\}$ exactly when $\mathsf{d}[i]\bigl(1-\mathsf{d}[i]\bigr)=0$, the check is a single \cref{piop:zerocheck} on the column $\mathsf{a}\cdot\mathsf{d}\cdot(1-\mathsf{d})$, which vanishes everywhere iff every active entry is boolean. When a protocol invokes the check on a bare data column, without naming an activator, the all-ones activator $\mathbf{1}$ is meant, so that the entry is boolean at \emph{every} point of the hypercube; this is the form we use for indicator columns that are later summed over the whole hypercube.
\begin{DefinePIOP}{Booleanity Check}{boolean-check}
  \emph{Inputs:} $\Oracle{\hat{\col}} = (\Oracle{\MLE{\mathsf{a}}}, \Oracle{\MLE{\mathsf{d}}})$.
  \begin{enumerate}[nosep]
    \item $\PIOPProver$ and $\PIOPVerifier$ invoke \cref{piop:zerocheck} on the column $\mathsf{a}\cdot\mathsf{d}\cdot(1-\mathsf{d})$.
  \end{enumerate}
\end{DefinePIOP}

\subsection{Limb Reconstruction Check}
The \emph{Limb Reconstruction Check} takes an $\alpha$-bit column $\col$ and $k$ columns $\col_0,\dots,\col_{k-1}$ each with $\alpha/k$ bits and proves that each entry of $\col$ is the concatenation of the corresponding entries in $\col_0,\dots,\col_{k-1}$. We package this as the relation
\ifsp
%% S&P (two-column): tuple and condition stacked so the display fits one column.
\begin{displaymath}
\NPRelation_{\mathsf{Limb}} = \left\{\,
  \begin{aligned}
    & \NPInstance = \alpha,\\
    & \NPInstancey = (\Oracle{\hat{\col}}, \Oracle{\hat{\col}_0}, \dots, \Oracle{\hat{\col}_{k-1}}),\\
    & \NPWitness = (\col, \col_0, \dots, \col_{k-1})\\[2pt]
    & \text{s.t.}\ \forall i,\, \col[i] = \sum_{j=0}^{k-1} 2^{(k-1-j)\alpha/k} \col_j[i]
  \end{aligned}
\,\right\}.
\end{displaymath}
\else
\begin{displaymath}
\NPRelation_{\mathsf{Limb}} = \left\{\,
  \NPInstance = \alpha,\;
  \NPInstancey = (\Oracle{\hat{\col}}, \Oracle{\hat{\col}_0}, \dots, \Oracle{\hat{\col}_{k-1}}),\;
  \NPWitness = (\col, \col_0, \dots, \col_{k-1})
  \;\middle|\;
    \forall i,\, \col[i] = \sum_{j=0}^{k-1} 2^{(k-1-j)\alpha/k} \col_j[i]
\,\right\}.
\end{displaymath}
\fi


Reconstruction is a single linear identity over the hypercube: subtracting the weighted limbs from $\col$ must vanish at every row. A single \cref{piop:zerocheck} on $\col - \sum_{j=0}^{k-1} 2^{(k-1-j)\alpha/k}\mult\col_j$ enforces exactly this, pinning down each entry of $\col$ as the concatenation of the corresponding limbs. The check certifies only the weighted-sum relationship; bounding each limb $\col_j$ to its $\alpha/k$-bit width, when required, is done by a separate \cref{piop:range-check}.
\begin{DefinePIOP}{Limb Reconstruction Check}{limb-check}
  \emph{Inputs:} $\Oracle{\hat{\col}}, \Oracle{\hat{\col}_0}, \ldots, \Oracle{\hat{\col}_{k-1}}$.
  \begin{enumerate}[nosep]
    \item $\PIOPProver$ and $\PIOPVerifier$ invoke \cref{piop:zerocheck} on the column $\col - \sum_{j=0}^{k-1} 2^{(k-1-j)\alpha/k}\mult\col_j$.
  \end{enumerate}
\end{DefinePIOP}

\subsection{Lookup Check}
The \emph{Lookup Check} takes two columns $\col_1 = (\mathsf{a}_1, \mathsf{d}_1)$ and $\col_2 = (\mathsf{a}_2, \mathsf{d}_2)$ and proves that every active value of $\col_1$ also occurs as an active value of $\col_2$, written $\col_1 \lookup \col_2$. We package this as the relation
\ifsp
%% S&P (two-column): tuple and condition stacked so the display fits one column.
\begin{displaymath}
\NPRelation_{\mathsf{Lookup}} = \left\{\,
  \begin{aligned}
    & \NPInstance = \bot,\\
    & \NPInstancey = (\Oracle{\hat{\col}_1}, \Oracle{\hat{\col}_2}),\\
    & \NPWitness = (\col_1, \col_2)\\[2pt]
    & \text{s.t.}\ \forall i,\, \mathsf{a}_1[i] = 1 \implies \exists j:\\
    & \phantom{\text{s.t.}}\ \quad \mathsf{a}_2[j] = 1 \wedge \mathsf{d}_1[i] = \mathsf{d}_2[j]
  \end{aligned}
\,\right\}.
\end{displaymath}
\else
\begin{displaymath}
\NPRelation_{\mathsf{Lookup}} = \left\{\,
  \NPInstance = \bot,\;
  \NPInstancey = (\Oracle{\hat{\col}_1}, \Oracle{\hat{\col}_2}),\;
  \NPWitness = (\col_1, \col_2)
  \;\middle|\;
    \forall i,\, \mathsf{a}_1[i] = 1 \implies \exists j:\, \mathsf{a}_2[j] = 1 \wedge \mathsf{d}_1[i] = \mathsf{d}_2[j]
\,\right\}.
\end{displaymath}
\fi
We discharge the lookup with the LogUp technique~\cite{Habock22}: $\col_1 \lookup \col_2$ holds if and only if there is a \emph{multiplicity} column $\mathsf{m}$ over $\col_2$'s hypercube such that, with $\mu$ and $\nu$ the hypercube dimensions of $\hat{\col}_1$ and $\hat{\col}_2$,
\begin{equation}
  \label{eq:lookup-logup}
  \sum_{x \in \BoolHCube{\mu}} \frac{\MLE{\mathsf{a}_1}(x)}{X - \MLE{\mathsf{d}_1}(x)}
  \;=\;
  \sum_{y \in \BoolHCube{\nu}} \frac{\MLE{\mathsf{a}_2}(y)\mult\MLE{\mathsf{m}}(y)}{X - \MLE{\mathsf{d}_2}(y)}
\end{equation}
as rational functions in $X$, where the activators zero out the inactive terms on each side. The prover sends $\mathsf{m}$, and the parties verify the rational identity by Schwartz--Zippel: the verifier samples a random $r \samples \F$, the prover sends the purported common value $s$ that both sides take at $X = r$, and the parties run \cref{piop:ratsumcheck} twice to confirm that each side evaluates to $s$ at $X = r$. When $\col_2$ is known to be duplicate-free, $\MLE{\mathsf{m}}(y)$ records the multiplicity of $\MLE{\mathsf{d}_2}(y)$ in $\col_1$, so the verifier stores $\Oracle{\MLE{\mathsf{m}}}$ as a protocol output for use by higher-level checks.
\begin{DefinePIOP}{Lookup Check}{lookup-check}
  \emph{Inputs:} $\Oracle{\hat{\col}_1}, \Oracle{\hat{\col}_2}$.
  \begin{enumerate}[nosep]
    \item $\PIOPProver$ constructs and sends the multiplicity oracle $\Oracle{\MLE{\mathsf{m}}}$, which $\PIOPVerifier$ stores as protocol output.
    \item $\PIOPVerifier$ samples a random point $r \samples \F$ and sends it to $\PIOPProver$, who responds with the claimed common value $s$ of both sides of \cref{eq:lookup-logup} at $X = r$.
    \item $\PIOPProver$ and $\PIOPVerifier$ invoke \cref{piop:ratsumcheck} twice to prove that both sides of \cref{eq:lookup-logup} evaluate to $s$ at $X = r$.
  \end{enumerate}
\end{DefinePIOP}


\subsection{Range Check}
The \emph{Range Check} takes a column $\col = (\mathsf{a}, \mathsf{d})$ and a width $\alpha$, and proves that every active entry of $\col$ is a non-negative integer of bitwidth $\alpha$, i.e.\ lies in $[0, 2^\alpha - 1]$. We package this as the relation
\ifsp
%% S&P (two-column): tuple and condition stacked so the display fits one column.
\begin{displaymath}
\NPRelation_{\mathsf{Range}} = \left\{\,
  \begin{aligned}
    & \NPInstance = \alpha,\\
    & \NPInstancey = (\Oracle{\hat{\col}}),\\
    & \NPWitness = (\col)\\[2pt]
    & \text{s.t.}\ \forall i,\, \mathsf{a}[i] = 1 \implies 0\leq \mathsf{d}[i] < 2^\alpha
  \end{aligned}
\,\right\}.
\end{displaymath}
\else
\begin{displaymath}
\NPRelation_{\mathsf{Range}} = \left\{\,
  \NPInstance = \alpha,\;
  \NPInstancey = (\Oracle{\hat{\col}}),\;
  \NPWitness = (\col)
  \;\middle|\;
    \forall i,\, \mathsf{a}[i] = 1 \implies 0\leq \mathsf{d}[i] < 2^\alpha
\,\right\}.
\end{displaymath}
\fi
We assume $\alpha < \log_2 |\F|$, so the range embeds unambiguously into $\F$. A direct approach proves $\col \lookup \mathsf{rng}_\alpha := (\mathbf{1}, \idx_\alpha)$, since the index polynomial $\idx_\alpha$ ranges over exactly $\{0, \ldots, 2^\alpha - 1\}$; but this lookup table has $2^\alpha$ rows and is wasteful for large $\alpha$. Instead we split each value into $16$-bit \emph{limbs} and range-check each limb against a small index column. Choosing $k$ and $\beta$ with $\alpha = 16k + \beta$ and $\beta \leq 16$, the prover sends limb columns $\mathsf{d}_1, \ldots, \mathsf{d}_{k+1}$ satisfying
\begin{displaymath}
  \MLE{\mathsf{a}}(x)\mult\MLE{\mathsf{d}}(x) = \sum_{i=1}^{k+1} 2^{16(i-1)}\mult\MLE{\mathsf{d}_i}(x), \qquad x \in \BoolHCube{\mu},
\end{displaymath}
which a single \cref{piop:zerocheck} enforces; the activator collapses inactive rows to zero, so the identity pins the limb decomposition on active rows. A \cref{piop:lookup-check} of each limb into the appropriate small index column then forces $\MLE{\mathsf{d}}(x) \in [0, 2^\alpha - 1]$ at every active $x$.
\begin{DefinePIOP}{Range Check}{range-check}
  \emph{Inputs:} $\Oracle{\hat{\col}} = (\Oracle{\MLE{\mathsf{a}}}, \Oracle{\MLE{\mathsf{d}}})$, width $\alpha$.
  \begin{enumerate}[nosep]
    \item $\PIOPProver$ and $\PIOPVerifier$ compute $\beta, k$ such that $16k + \beta = \alpha$ and $\beta \leq 16$.
    \item $\PIOPProver$ constructs and sends the limb oracles $\Oracle{\MLE{\mathsf{d}_1}}, \ldots, \Oracle{\MLE{\mathsf{d}_{k+1}}}$, the $16$-bit limb decomposition of $\mathsf{d}$.
    \item $\PIOPProver$ and $\PIOPVerifier$ invoke \cref{piop:zerocheck} on the column $\mathsf{a}\mult\mathsf{d} - \sum_{i=1}^{k+1} 2^{16(i-1)}\mult\mathsf{d}_i$.
    \item For each $i = 1, \dots, k$, $\PIOPProver$ and $\PIOPVerifier$ invoke \cref{piop:lookup-check} on $(\mathbf{1}, \mathsf{d}_i)$ and $\mathsf{rng}_{16} := (\mathbf{1}, \idx_{16})$, proving each limb lies in $[0, 2^{16}-1]$.
    \item $\PIOPProver$ and $\PIOPVerifier$ invoke \cref{piop:lookup-check} on $(\mathbf{1}, \mathsf{d}_{k+1})$ and $\mathsf{rng}_\beta := (\mathbf{1}, \idx_\beta)$, proving the residual limb lies in $[0, 2^\beta-1]$.
  \end{enumerate}
\end{DefinePIOP}

\subsection{Sign Check}
\label{sec:sign-check}
The \emph{Sign Check} takes a column $\col = (\mathsf{a}, \mathsf{d})$, a width $\alpha$, and a sign mode $\sigma \in \{\mathsf{NonNeg}, \mathsf{Pos}, \mathsf{NonPos}, \mathsf{Neg}\}$, and proves that every active entry of $\col$ lies in the width-$\alpha$ slice of the corresponding sign. Each mode is determined by two parameters: a \emph{direction} $s_\sigma \in \{+1, -1\}$, which is $+1$ for the positive-side modes $\mathsf{NonNeg}, \mathsf{Pos}$ and $-1$ for the negative-side modes $\mathsf{NonPos}, \mathsf{Neg}$; and a lower bound $\ell_\sigma \in \{0, 1\}$, which is $0$ for the non-strict modes $\mathsf{NonNeg}, \mathsf{NonPos}$ and $1$ for the strict modes $\mathsf{Pos}, \mathsf{Neg}$ that exclude zero. Concretely,
\begin{center}\small
\begin{tabular}{lcccc}
\toprule
$\sigma$ & $\mathsf{NonNeg}$ & $\mathsf{Pos}$ & $\mathsf{NonPos}$ & $\mathsf{Neg}$ \\
\midrule
certified range & $[0, 2^\alpha-1]$ & $[1, 2^\alpha-1]$ & $[-(2^\alpha-1), 0]$ & $[-(2^\alpha-1), -1]$ \\
$(s_\sigma, \ell_\sigma)$ & $(+1, 0)$ & $(+1, 1)$ & $(-1, 0)$ & $(-1, 1)$ \\
\bottomrule
\end{tabular}
\end{center}
where negative values are taken in $\F$ via the additive inverse. In every mode the certified condition is $s_\sigma\mult\mathsf{d}[i] \in [\ell_\sigma, 2^\alpha-1]$, so we package the check as the relation
\ifsp
%% S&P (two-column): tuple and condition stacked so the display fits one column.
\begin{displaymath}
\NPRelation_{\mathsf{Sign}} = \left\{\,
  \begin{aligned}
    & \NPInstance = (\alpha, \sigma),\\
    & \NPInstancey = (\Oracle{\hat{\col}}),\\
    & \NPWitness = (\col)\\[2pt]
    & \text{s.t.}\ \forall i,\, \mathsf{a}[i] = 1 \implies{}\\
    & \phantom{\text{s.t.}}\ \quad \ell_\sigma \leq s_\sigma\mult\mathsf{d}[i] \leq 2^\alpha - 1
  \end{aligned}
\,\right\}.
\end{displaymath}
\else
\begin{displaymath}
\NPRelation_{\mathsf{Sign}} = \left\{\,
  \NPInstance = (\alpha, \sigma),\;
  \NPInstancey = (\Oracle{\hat{\col}}),\;
  \NPWitness = (\col)
  \;\middle|\;
    \forall i,\, \mathsf{a}[i] = 1 \implies \ell_\sigma \leq s_\sigma\mult\mathsf{d}[i] \leq 2^\alpha - 1
\,\right\}.
\end{displaymath}
\fi
We assume $2^{\alpha+1} < |\F|$, so that the range embeds unambiguously into $\F$ and, as the indicator checks below require, the positive-side range $[0, 2^\alpha-1]$ and the negative-side range $[-(2^\alpha-1), -1]$ are disjoint subsets of $\F$.

The whole check reduces to a non-negativity check on the \emph{magnitude} column $\mathsf{m} := s_\sigma\mult\mathsf{d}$, which the verifier obtains by negating the oracle $\MLE{\mathsf{d}}$ in the negative-side modes. The index polynomial $\idx_\alpha$, evaluated over $\BoolHCube{\alpha}$, takes exactly the values $\{0, 1, \ldots, 2^\alpha - 1\}$. Treating it as the data of an $\alpha$-variable column with the all-ones activator gives a range column $\mathsf{rng}_\alpha := (\mathbf{1}, \idx_\alpha)$ whose value multiset is precisely $[0, 2^\alpha - 1]$, so a single \cref{piop:lookup-check} of $\mathsf{m}$ into $\mathsf{rng}_\alpha$ is already an exact range check. This naive lookup is wasteful when $\alpha$ is large, since $\mathsf{rng}_\alpha$ has $2^\alpha$ rows. We do better by splitting each entry into $16$-bit \emph{limbs} and range-checking each limb against a small index column. Choosing $k$ and $\beta$ with $\alpha = 16k + \beta$ and $\beta \leq 16$, the prover sends limb columns $\mathsf{m}_1, \ldots, \mathsf{m}_{k+1}$ satisfying
\begin{displaymath}
  \MLE{\mathsf{a}}(x) \mult s_\sigma \mult \MLE{\mathsf{d}}(x) = \sum_{i=1}^{k+1} 2^{16(i-1)} \mult \MLE{\mathsf{m}_i}(x), \qquad x \in \BoolHCube{\mu},
\end{displaymath}
which a single \cref{piop:zerocheck} enforces; the activator on the left collapses inactive rows to zero, so the identity pins down the limb decomposition on active rows and forces the limbs to combine to zero on inactive rows. Range-checking each limb then pins $s_\sigma\mult\MLE{\mathsf{d}}(x) \in [0, 2^\alpha - 1]$ at every active $x$. In the strict modes $\mathsf{Pos}$ and $\mathsf{Neg}$, a final no-zero check rules out the value $0$, narrowing the certified range to $[1, 2^\alpha - 1]$. When convenient we write $\col \geq 0$, $\col > 0$, $\col \leq 0$ and $\col < 0$ for the Sign Check on $\col$ in the modes $\mathsf{NonNeg}$, $\mathsf{Pos}$, $\mathsf{NonPos}$ and $\mathsf{Neg}$ respectively, with the width given by the context.
\begin{DefinePIOP}{Sign Check}{sign-check}
  \emph{Inputs:} $\Oracle{\hat{\col}} = (\Oracle{\MLE{\mathsf{a}}}, \Oracle{\MLE{\mathsf{d}}})$, width $\alpha$, sign mode $\sigma \in \{\mathsf{NonNeg}, \mathsf{Pos}, \mathsf{NonPos}, \mathsf{Neg}\}$.
  \begin{enumerate}[nosep]
    \item $\PIOPProver$ and $\PIOPVerifier$ read off $s_\sigma \in \{+1,-1\}$ and $\ell_\sigma \in \{0,1\}$ from $\sigma$, and compute $\beta, k$ such that $16k + \beta = \alpha$ and $\beta \leq 16$.
    \item $\PIOPProver$ constructs and sends the limb oracles $\Oracle{\MLE{\mathsf{m}_1}}, \ldots, \Oracle{\MLE{\mathsf{m}_{k+1}}}$ of the magnitude $\mathsf{m} = s_\sigma\mult\mathsf{d}$.
    \item $\PIOPProver$ and $\PIOPVerifier$ invoke \cref{piop:zerocheck} to prove $\mathsf{a}\mult s_\sigma\mult\mathsf{d} - \sum_{i=1}^{k+1} 2^{16(i-1)}\mult\mathsf{m}_i = 0$.
    \item For each $i = 1, \dots, k$, $\PIOPProver$ and $\PIOPVerifier$ invoke \cref{piop:lookup-check} on $(\mathbf{1}, \mathsf{m}_i)$ and $\mathsf{rng}_{16} := (\mathbf{1}, \idx_{16})$, proving each limb lies in $[0, 2^{16}-1]$.
    \item $\PIOPProver$ and $\PIOPVerifier$ invoke \cref{piop:lookup-check} on $(\mathbf{1}, \mathsf{m}_{k+1})$ and $\mathsf{rng}_\beta := (\mathbf{1}, \idx_\beta)$, proving the residual limb lies in $[0, 2^\beta-1]$.
    \item If $\ell_\sigma = 1$ (the strict modes $\mathsf{Pos}, \mathsf{Neg}$), $\PIOPProver$ and $\PIOPVerifier$ additionally invoke \cref{piop:nozero-check} on $\col$ to rule out zero entries, narrowing the certified range to $[1, 2^\alpha - 1]$.
  \end{enumerate}
\end{DefinePIOP}

\subsection{Zero Indicator Check}
\label{sec:zero-indicator}
Several protocols need a boolean column that records \emph{where} a column vanishes, rather than an assertion that it does. The \emph{Zero Indicator Check} takes a column $\col = (\mathsf{a},\mathsf{d})$ and a column $\mathsf{z}$ sent by the prover, and proves that on the active rows $\mathsf{z}$ is boolean and marks exactly the zero entries of $\col$. We package this as the relation
\ifsp
\begin{displaymath}
\NPRelation_{\mathsf{ZeroInd}} = \left\{\,
  \begin{aligned}
    & \NPInstance = \bot,\\
    & \NPInstancey = (\Oracle{\hat{\col}}, \Oracle{\MLE{\mathsf{z}}}),\\
    & \NPWitness = (\col, \mathsf{z})\\[2pt]
    & \text{s.t.}\ \forall i,\, \mathsf{a}[i] = 1 \implies \mathsf{z}[i] = \mathbf{1}[\mathsf{d}[i] = 0]
  \end{aligned}
\,\right\}.
\end{displaymath}
\else
\begin{displaymath}
\NPRelation_{\mathsf{ZeroInd}} = \left\{\,
  \NPInstance = \bot,\;
  \NPInstancey = (\Oracle{\hat{\col}}, \Oracle{\MLE{\mathsf{z}}}),\;
  \NPWitness = (\col, \mathsf{z})
  \;\middle|\;
    \forall i,\, \mathsf{a}[i] = 1 \implies \mathsf{z}[i] = \mathbf{1}[\mathsf{d}[i] = 0]
\,\right\}.
\end{displaymath}
\fi
The check is the conjunction of three facts: $\mathsf{z}$ is boolean on the active rows; wherever $\mathsf{z}$ is $1$ the data vanishes, which is a zerocheck on $\mathsf{a}\mult\mathsf{z}\mult\mathsf{d}$; and wherever $\mathsf{z}$ is $0$ the data is nonzero, which is a no-zero check on $\mathsf{d}$ under the activator $\mathsf{a}\mult(1-\mathsf{z})$. Together they leave the prover no freedom in $\mathsf{z}$ on the active rows.
\begin{DefinePIOP}{Zero Indicator Check}{zero-indicator}
  \emph{Inputs:} $\Oracle{\hat{\col}} = (\Oracle{\MLE{\mathsf{a}}}, \Oracle{\MLE{\mathsf{d}}})$ and the indicator $\Oracle{\MLE{\mathsf{z}}}$, sent by $\PIOPProver$ with $\mathsf{z}[i] = \mathbf{1}[\mathsf{d}[i] = 0]$ at active indices.
  \begin{enumerate}[nosep]
    \item $\PIOPProver$ and $\PIOPVerifier$ invoke \cref{piop:boolean-check} on the column with activator $\mathsf{a}$ and data $\mathsf{z}$.
    \item $\PIOPProver$ and $\PIOPVerifier$ invoke \cref{piop:zerocheck} on $\mathsf{a}\mult\mathsf{z}\mult\mathsf{d}$.
    \item $\PIOPProver$ and $\PIOPVerifier$ invoke \cref{piop:nozero-check} on the column with activator $\mathsf{a}\mult(1-\mathsf{z})$ and data $\mathsf{d}$.
  \end{enumerate}
\end{DefinePIOP}

\subsection{Comparison Indicator Check}
\label{sec:cmp-indicator}
The Sign Check asserts the sign of every active entry of a column. The protocols of \cref{sec:piop-for-pattern-matching} instead need a boolean column that \emph{records} the sign of each entry, so that it can be multiplied into activators. The \emph{Comparison Indicator Check} takes a column $\col = (\mathsf{a},\mathsf{d})$ whose active entries are signed integers of magnitude below $2^\alpha$ (negative values taken in $\F$ via the additive inverse, as in \cref{sec:sign-check}), and a column $\mathsf{b}$ sent by the prover, and proves that on the active rows $\mathsf{b}$ is boolean and marks exactly the non-negative entries. We package this as the relation
\ifsp
\begin{displaymath}
\NPRelation_{\mathsf{CmpInd}} = \left\{\,
  \begin{aligned}
    & \NPInstance = \alpha,\\
    & \NPInstancey = (\Oracle{\hat{\col}}, \Oracle{\MLE{\mathsf{b}}}),\\
    & \NPWitness = (\col, \mathsf{b})\\[2pt]
    & \text{s.t.}\ \forall i,\, \mathsf{a}[i] = 1 \implies{}\\
    & \phantom{\text{s.t.}}\ \quad \mathsf{d}[i] \in [-(2^\alpha-1), 2^\alpha-1]\ \text{and}\\
    & \phantom{\text{s.t.}}\ \quad \mathsf{b}[i] = \mathbf{1}[\mathsf{d}[i] \ge 0]
  \end{aligned}
\,\right\}.
\end{displaymath}
\else
\begin{displaymath}
\NPRelation_{\mathsf{CmpInd}} = \left\{\,
  \NPInstance = \alpha,\;
  \NPInstancey = (\Oracle{\hat{\col}}, \Oracle{\MLE{\mathsf{b}}}),\;
  \NPWitness = (\col, \mathsf{b})
  \;\middle|\;
    \forall i,\, \mathsf{a}[i] = 1 \implies \mathsf{d}[i] \in [-(2^\alpha-1), 2^\alpha-1]\ \wedge\ \mathsf{b}[i] = \mathbf{1}[\mathsf{d}[i] \ge 0]
\,\right\}.
\end{displaymath}
\fi
The check splits the active rows according to $\mathsf{b}$ and runs a Sign Check on each part: the rows with $\mathsf{b} = 1$ must be $\mathsf{NonNeg}$ and the rows with $\mathsf{b} = 0$ must be $\mathsf{Neg}$. Since $2^{\alpha+1} < |\F|$, the two certified ranges $[0, 2^\alpha-1]$ and $[-(2^\alpha-1), -1]$ are disjoint in $\F$, so no entry can be placed on the wrong side, and an entry outside both ranges fails whichever side it is placed on. We write $\mathbf{1}[\col \ge 0]$ for the certified indicator; comparisons between two columns are reduced to this form through their difference, e.g.\ $\mathbf{1}[\mathsf{u} \ge \mathsf{v}] = \mathbf{1}[\mathsf{u} - \mathsf{v} \ge 0]$ and $\mathbf{1}[\mathsf{u} < \mathsf{v}] = 1 - \mathbf{1}[\mathsf{u} - \mathsf{v} \ge 0]$.
\begin{DefinePIOP}{Comparison Indicator Check}{cmp-indicator}
  \emph{Inputs:} $\Oracle{\hat{\col}} = (\Oracle{\MLE{\mathsf{a}}}, \Oracle{\MLE{\mathsf{d}}})$, width $\alpha$, and the indicator $\Oracle{\MLE{\mathsf{b}}}$, sent by $\PIOPProver$ with $\mathsf{b}[i] = \mathbf{1}[\mathsf{d}[i] \ge 0]$ at active indices.
  \begin{enumerate}[nosep]
    \item $\PIOPProver$ and $\PIOPVerifier$ invoke \cref{piop:boolean-check} on the column with activator $\mathsf{a}$ and data $\mathsf{b}$.
    \item $\PIOPProver$ and $\PIOPVerifier$ invoke the $\mathsf{NonNeg}$ \cref{piop:sign-check} with width $\alpha$ on the column with activator $\mathsf{a}\mult\mathsf{b}$ and data $\mathsf{d}$.
    \item $\PIOPProver$ and $\PIOPVerifier$ invoke the $\mathsf{Neg}$ \cref{piop:sign-check} with width $\alpha$ on the column with activator $\mathsf{a}\mult(1-\mathsf{b})$ and data $\mathsf{d}$.
  \end{enumerate}
\end{DefinePIOP}

\subsection{Keyed Sumcheck}
The \emph{Keyed Sumcheck} takes two tables $\Left$ and $\Right$, each with a key column $\key$, a value column $\val$, and an activator column $\act$, and proves that for every possible key $\elem \in \F$, the sum of active values in $\Left$ with key $\elem$ equals the sum of active values in $\Right$ with key $\elem$. Define the keyed sums
\ifsp
%% S&P (two-column): the two keyed sums stacked so the display fits one column.
\begin{align*}
    S_\Left(\elem) &:= \sum_{\substack{i \in [|\Left|] \\ \Left.\key[i] = \elem}} \Left.\act[i]\mult\Left.\val[i]
    \quad\text{and}\\
    S_\Right(\elem) &:= \sum_{\substack{j \in [|\Right|] \\ \Right.\key[j] = \elem}} \Right.\act[j]\mult\Right.\val[j].
\end{align*}
\else
\begin{displaymath}
    S_\Left(\elem) := \sum_{\substack{i \in [|\Left|] \\ \Left.\key[i] = \elem}} \Left.\act[i]\mult\Left.\val[i]
    \qquad\text{and}\qquad
    S_\Right(\elem) := \sum_{\substack{j \in [|\Right|] \\ \Right.\key[j] = \elem}} \Right.\act[j]\mult\Right.\val[j].
\end{displaymath}
\fi
We package this as the relation
\ifsp
%% S&P (two-column): tuple and condition stacked so the display fits one column.
\begin{displaymath}
\NPRelation_{\mathsf{ksum}} = \left\{\,
  \begin{aligned}
    & \NPInstance = \bot,\\
    & \NPInstancey = (\Oracle{\hat{\Left}}, \Oracle{\hat{\Right}}),\\
    & \NPWitness = (\Left, \Right)\\[2pt]
    & \text{s.t.}\ \forall\, \elem \in \F : S_\Left(\elem) = S_\Right(\elem)
  \end{aligned}
\,\right\}.
\end{displaymath}
\else
\begin{displaymath}
\NPRelation_{\mathsf{ksum}} = \left\{\,
  \NPInstance = \bot,\;
  \NPInstancey = (\Oracle{\hat{\Left}}, \Oracle{\hat{\Right}}),\;
  \NPWitness = (\Left, \Right)
  \;\middle|\;
    \forall\, \elem \in \F : S_\Left(\elem) = S_\Right(\elem)
\,\right\}.
\end{displaymath}
\fi

\cref{piop:keyedsumcheck} reduces the per-key equality $\forall \elem : S_\Left(\elem) = S_\Right(\elem)$ to a single rational identity. The protocol is a generalization of the LogUp lookup protocol~\cite{Habock22}, and we derive its security from a core lemma about function equality that is implicit in LogUp.
\begin{ilemma}[function equality~\cite{Habock22}]
    \label{lemma:func-equality}
    For any functions $S_1, S_2: \F \to \F$, $S_1(\elem) = S_2(\elem)$ for every $\elem \in \F$ if and only if
    \begin{displaymath}
        \sum_{\elem \in \F} \frac{S_1(\elem)}{X - \elem} = \sum_{\elem \in \F} \frac{S_2(\elem)}{X - \elem}
    \end{displaymath}
    as rational functions in the formal variable $X$.
\end{ilemma}
Applying Lemma~\ref{lemma:func-equality} to $S_\Left$ and $S_\Right$ and collapsing the per-key double sums into single sums over rows (the per-key index sets partition the rows), the property holds if and only if, with $\mu$ and $\nu$ the hypercube dimensions of $\hat{\Left}$ and $\hat{\Right}$,
\begin{equation}
    \label{eq:keyedsumcheck_rational}
    \sum_{x \in \BoolHCube{\mu}} \frac{\hat{\Left}.\act(x) \mult \hat{\Left}.\val(x)}{X - \hat{\Left}.\key(x)}
    \;=\;
    \sum_{x \in \BoolHCube{\nu}} \frac{\hat{\Right}.\act(x) \mult \hat{\Right}.\val(x)}{X - \hat{\Right}.\key(x)}.
\end{equation}
We verify this rational identity by Schwartz--Zippel: the verifier samples a random $r \samples \F$, the prover sends the purported common value $s$ that both sides take at $X = r$, and the parties run \cref{piop:ratsumcheck} twice to confirm that both sides actually evaluate to $s$ at $X = r$.
\begin{DefinePIOP}{KeyedSumcheck}{keyedsumcheck}
  \emph{Inputs:} $\Oracle{\hat{\Left}}, \Oracle{\hat{\Right}}$.
    \begin{enumerate}[nosep]
        \item $\PIOPVerifier$ samples a random point $r \samples \F$ and sends it to $\PIOPProver$.
        \item $\PIOPProver$ computes and sends $\PIOPVerifier$ the value $s$, the purported evaluation of both sides of \cref{eq:keyedsumcheck_rational} at $X = r$.
        \item $\PIOPProver$ and $\PIOPVerifier$ invoke \cref{piop:ratsumcheck} twice to prove that both sides of \cref{eq:keyedsumcheck_rational} evaluate to $s$ at $X = r$.
    \end{enumerate}
\end{DefinePIOP}

\subsection{Multiset Equality Check}
The \emph{Multiset Equality Check} takes two columns $\col_\Left$ and $\col_\Right$ and proves that they hold the same multiset of active values. We package this as the relation
\ifsp
%% S&P (two-column): tuple and condition stacked so the display fits one column.
\begin{displaymath}
\NPRelation_{\mathsf{mset}} = \left\{\,
  \begin{aligned}
    & \NPInstance = \bot,\\
    & \NPInstancey = (\Oracle{\hat{\col}_\Left}, \Oracle{\hat{\col}_\Right}),\\
    & \NPWitness = (\col_\Left, \col_\Right)\\[2pt]
    & \text{s.t.}\ \{\!\!\{\, \col_\Left[i] \,\}\!\!\}_i = \{\!\!\{\, \col_\Right[j] \,\}\!\!\}_j
  \end{aligned}
\,\right\}.
\end{displaymath}
\else
\begin{displaymath}
\NPRelation_{\mathsf{mset}} = \left\{\,
  \NPInstance = \bot,\;
  \NPInstancey = (\Oracle{\hat{\col}_\Left}, \Oracle{\hat{\col}_\Right}),\;
  \NPWitness = (\col_\Left, \col_\Right)
  \;\middle|\;
  \{\!\!\{\, \col_\Left[i] \,\}\!\!\}_i = \{\!\!\{\, \col_\Right[j] \,\}\!\!\}_j
\,\right\}.
\end{displaymath}
\fi
Two multisets agree exactly when every value $\elem \in \F$ occurs with the same multiplicity on both sides, i.e.\ $\forall\, \elem \in \F : \sum_{i:\,\col_\Left[i]=\elem} 1 = \sum_{j:\,\col_\Right[j]=\elem} 1$. This is precisely an $\elem$-keyed sum equality between two tables whose key is the column data and whose value is the constant $\mathbf{1}$. The parties locally form the auxiliary tables
\ifsp
%% S&P (two-column): the two tables stacked so the display fits one column.
\begin{align*}
    \Left &:= \bigl(\act = \col_\Left.\act,\; \key = \col_\Left.\data,\; \val = \mathbf{1}\bigr)
    \quad\text{and}\\
    \Right &:= \bigl(\act = \col_\Right.\act,\; \key = \col_\Right.\data,\; \val = \mathbf{1}\bigr),
\end{align*}
\else
\begin{displaymath}
    \Left := \bigl(\act = \col_\Left.\act,\; \key = \col_\Left.\data,\; \val = \mathbf{1}\bigr)
    \quad\text{and}\quad
    \Right := \bigl(\act = \col_\Right.\act,\; \key = \col_\Right.\data,\; \val = \mathbf{1}\bigr),
\end{displaymath}
\fi
and a single \cref{piop:keyedsumcheck} on $\Left$ and $\Right$ closes out the proof, with completeness and soundness inherited from the keyed sumcheck.
\begin{DefinePIOP}{Multiset Equality Check}{mset}
  \emph{Inputs:} $\Oracle{\hat{\col}_\Left}, \Oracle{\hat{\col}_\Right}$.
  \begin{enumerate}[nosep]
    \item $\PIOPProver$ and $\PIOPVerifier$ locally form the tables $\Left$ and $\Right$ as above.
    \item $\PIOPProver$ and $\PIOPVerifier$ invoke \cref{piop:keyedsumcheck} on $\Left$ and $\Right$.
  \end{enumerate}
\end{DefinePIOP}

\parhead{Handling tables} The foregoing column protocol extends to prove that two \emph{tables} hold the same multiset of \emph{rows}: the parties first fingerprint each table to a single column with a verifier-chosen random linear combination of its data columns, and then invoke the foregoing check on the two fingerprint columns.

\subsection{Rotation Check}
The \emph{Rotation Check} takes two data vectors $\mathsf{d}$ and $\mathsf{d}'$ over $\BoolHCube{\mu}$, a shift $t \in \mathbb{Z}$ and a public length $s \le 2^\mu$, and proves that on the first $s$ positions $\mathsf{d}'$ is the cyclic rotation of $\mathsf{d}$ by $t$ positions, written $\mathsf{d}' = \rho_t(\mathsf{d})$, where $\rho_t$ sends the entry at index $p$ to index $p+t$ modulo $s$. We package this as the relation
\ifsp
%% S&P (two-column): tuple and condition stacked so the display fits one column.
\begin{displaymath}
\NPRelation_{\mathsf{rot}} = \left\{\,
  \begin{aligned}
    & \NPInstance = (t, s),\\
    & \NPInstancey = (\Oracle{\MLE{\mathsf{d}}}, \Oracle{\MLE{\mathsf{d}'}}),\\
    & \NPWitness = (\mathsf{d}, \mathsf{d}')\\[2pt]
    & \text{s.t.}\ \forall\, q \in [s] : \mathsf{d}'[q] = \mathsf{d}[(q - t) \bmod s]
  \end{aligned}
\,\right\}.
\end{displaymath}
\else
\begin{displaymath}
\NPRelation_{\mathsf{rot}} = \left\{\,
  \NPInstance = (t, s),\;
  \NPInstancey = (\Oracle{\MLE{\mathsf{d}}}, \Oracle{\MLE{\mathsf{d}'}}),\;
  \NPWitness = (\mathsf{d}, \mathsf{d}')
  \;\middle|\;
    \forall\, q \in [s] : \mathsf{d}'[q] = \mathsf{d}[(q - t) \bmod s]
\,\right\}.
\end{displaymath}
\fi
The length $s$ is part of the instance and is never chosen by the prover. For a column in the contiguous layout it is the active length $n$, which the calling protocol has already fixed by asserting that the activator is $\contig_{\mu,n}$, and we omit it from such invocations; for the char-level columns of \cref{sec:piop-for-pattern-matching}, whose activators are sparse, it is the full hypercube, $s = 2^\mu$, so that every position of $\mathsf{d}'$ is constrained. The relation says nothing about the positions $q \ge s$ of $\mathsf{d}'$, and a calling protocol must not read them.

We reduce the rotation condition to a single multiset equality on \emph{position-tagged} columns: tag each entry of $\mathsf{d}$ with its own position, and tag each entry of $\mathsf{d}'$ with the position it should occupy in $\mathsf{d}$, restricting both to the first $s$ positions through the activator $\contig_{\mu,s}$, which the verifier evaluates on its own. Concretely, the parties locally construct the tagged tables
\begin{displaymath}
  \Table := \bigl(\contig_{\mu,s},\; \mathsf{d},\; \idx_\mu\bigr)
  \quad\text{and}\quad
  \Table' := \bigl(\contig_{\mu,s},\; \mathsf{d}',\; \idxrot\bigr),
\end{displaymath}
where $\idx_\mu$ maps each $x \in \BoolHCube{\mu}$ to its hypercube position and $\idxrot$ is the \emph{shifted} index polynomial sending each position $q < s$ to $(q - t) \bmod s$. Writing $t_0 := t \bmod s$, this shift is realized exactly by
\begin{displaymath}
  \idxrot(x) := \idx_\mu(x) - t_0 + s\mult\contig_{\mu, t_0}(x),
\end{displaymath}
since $\contig_{\mu, t_0}(x) = 1$ precisely when $\idx_\mu(x) < t_0$, supplying the $+s$ wraparound exactly on the positions that underflow. Because $q \mapsto (q - t) \bmod s$ is a bijection on $\{0, \ldots, s-1\}$, the tagged tables agree as multisets of rows if and only if $\mathsf{d}'[q] = \mathsf{d}[(q-t) \bmod s]$ for every $q < s$, which is exactly the rotation condition. A single \cref{piop:mset} on $\Table$ and $\Table'$ then proves the claim, with completeness and soundness inherited from the multiset equality check.
\begin{DefinePIOP}{Rotation Check}{rotcheck}
  \emph{Inputs:} $\Oracle{\MLE{\mathsf{d}}}, \Oracle{\MLE{\mathsf{d}'}}$, the shift $t$, and the length $s$.
  \begin{enumerate}[nosep]
    \item $\PIOPProver$ and $\PIOPVerifier$ locally construct the tagged tables $\Table$ and $\Table'$ as above, with activator $\contig_{\mu,s}$ and the shifted index polynomial $\idxrot(x) = \idx_\mu(x) - t_0 + s\mult\contig_{\mu, t_0}(x)$ with $t_0 := t \bmod s$.
    \item $\PIOPProver$ and $\PIOPVerifier$ invoke \cref{piop:mset} to prove that $\Table$ and $\Table'$ are equal as multisets of rows.
  \end{enumerate}
\end{DefinePIOP}


\subsection{Lexicographic Sort Check}
\label{sec:lex-sort}
The \emph{Lexicographic Sort Check} is the multi-key sortedness protocol of $\truthtable$~\cite{cryptoeprint:2026/1260}, which we restate in our notation. It takes a table $\Table = (\mathsf{a}, [\mathsf{d}_1, \dots, \mathsf{d}_m])$ in contiguous layout with $n$ active rows, together with a boolean \emph{initial tie indicator} $\mathsf{t}_0$ marking a subset of the adjacent row pairs $(j, j+1)$, and proves that every marked pair is in non-decreasing lexicographic order of the keys, $(\mathsf{d}_1[j], \dots, \mathsf{d}_m[j]) \le_{\mathsf{lex}} (\mathsf{d}_1[j+1], \dots, \mathsf{d}_m[j+1])$. In \emph{strict} mode it proves in addition that no marked pair is equal on all $m$ keys. We package this as the relation
\ifsp
\begin{displaymath}
\NPRelation_{\mathsf{LexSort}} = \left\{\,
  \begin{aligned}
    & \NPInstance = (m, \mathsf{strict}),\\
    & \NPInstancey = (\Oracle{\hat{\Table}}, \Oracle{\MLE{\mathsf{t}_0}}),\\
    & \NPWitness = (\Table, \mathsf{t}_0)\\[2pt]
    & \text{s.t.}\ \mathsf{a} = \contig_{\mu,n},\ \text{and}\ \forall j\ \text{with}\ \mathsf{t}_0[j] = 1:\\
    & \phantom{\text{s.t.}}\ (\mathsf{d}_1[j], \dots, \mathsf{d}_m[j]) \le_{\mathsf{lex}} (\mathsf{d}_1[j+1], \dots, \mathsf{d}_m[j+1]),\\
    & \phantom{\text{s.t.}}\ \text{with}\ \ne\ \text{if}\ \mathsf{strict}
  \end{aligned}
\,\right\}.
\end{displaymath}
\else
\begin{displaymath}
\NPRelation_{\mathsf{LexSort}} = \left\{\,
  \NPInstance = (m, \mathsf{strict}),\;
  \NPInstancey = (\Oracle{\hat{\Table}}, \Oracle{\MLE{\mathsf{t}_0}}),\;
  \NPWitness = (\Table, \mathsf{t}_0)
  \;\middle|\;
  \begin{array}{l}
    \mathsf{a} = \contig_{\mu,n},\ \text{and for all } j \text{ with } \mathsf{t}_0[j] = 1: \\[3pt]
    (\mathsf{d}_1[j], \dots, \mathsf{d}_m[j]) \le_{\mathsf{lex}} (\mathsf{d}_1[j+1], \dots, \mathsf{d}_m[j+1]),\ \text{with} \ne \text{ if } \mathsf{strict}
  \end{array}
\,\right\}.
\end{displaymath}
\fi
The protocol mirrors the way one sorts by several keys: order by the first key, then order the runs of rows tied on the first key by the second key, and so on. For each key $i$ the prover sends a \emph{tie indicator} $\mathsf{t}_i$ marking the adjacent pairs that are tied on keys $1, \dots, i$, together with the shifted copy $\mathsf{d}_i^{\mathsf{next}} := \rho_{-1}(\mathsf{d}_i)$ that aligns every row with its successor. Three checks per key certify the bookkeeping and the order: ties can only persist ($\mathsf{t}_i \le \mathsf{t}_{i-1}$); a persisting tie is a genuine tie ($\mathsf{d}_i[j] = \mathsf{d}_i[j+1]$ wherever $\mathsf{t}_i[j] = 1$); and a tie broken at key $i$ is broken in the right direction ($\mathsf{d}_i[j] < \mathsf{d}_i[j+1]$ wherever $\mathsf{t}_{i-1}[j] = 1$ and $\mathsf{t}_i[j] = 0$). Together these force $\mathsf{t}_i = \mathsf{t}_{i-1} \mult \mathbf{1}[\mathsf{d}_i[j] = \mathsf{d}_i[j+1]]$ and $\mathsf{d}_i[j] \le \mathsf{d}_i[j+1]$ on every pair of $\mathsf{t}_{i-1}$, which is the lexicographic condition by induction on $i$. Our restatement differs from $\truthtable$'s in two cosmetic ways that later sections rely on: the initial tie indicator $\mathsf{t}_0$ is an input rather than fixed to $\contig_{\mu,n-1}$ (all adjacent pairs, the case of a plain \texttt{ORDER BY}), and the final tie indicator $\mathsf{t}_m$ is returned, so that several invocations can be chained over keys of different kinds.
\begin{DefinePIOP}{Lexicographic Sort Check}{lex-sort}
  \emergencystretch=3em\relax
  \emph{Inputs:} $\Oracle{\MLE{\mathsf{a}}}, \Oracle{\MLE{\mathsf{d}_1}}, \dots, \Oracle{\MLE{\mathsf{d}_m}}$; the initial tie indicator $\Oracle{\MLE{\mathsf{t}_0}}$ (by default $\contig_{\mu,n-1}$); a width $\alpha$ bounding $|\mathsf{d}_i[j] - \mathsf{d}_i[j+1]|$; and the flag $\mathsf{strict}$.
  \vspace{1mm}

  \noindent\emph{Output:} the final tie indicator $\Oracle{\MLE{\mathsf{t}_m}}$.
  \vspace{2mm}
  \begin{enumerate}[nosep]
    \item $\PIOPProver$ sends the active length $n$, and $\PIOPVerifier$ asserts $\mathsf{a} = \contig_{\mu,n}$.
    \item For $i = 1, \dots, m$, $\PIOPProver$ sends $\Oracle{\MLE{\mathsf{d}_i^{\mathsf{next}}}}$ and $\Oracle{\MLE{\mathsf{t}_i}}$ with $\mathsf{d}_i^{\mathsf{next}}[j] := \mathsf{d}_i[j+1]$ and $\mathsf{t}_i[j] := \mathsf{t}_{i-1}[j] \mult \mathbf{1}[\mathsf{d}_i[j] = \mathsf{d}_i[j+1]]$, and $\PIOPProver$ and $\PIOPVerifier$ invoke
    \begin{enumerate}[nosep]
      \item \cref{piop:rotcheck} on $(\mathsf{d}_i, \mathsf{d}_i^{\mathsf{next}}, -1)$;
      \item \cref{piop:boolean-check} on $\mathsf{t}_i$ and \cref{piop:zerocheck} on $\mathsf{t}_i \mult (1 - \mathsf{t}_{i-1})$;
      \item (\emph{persisting ties}) \cref{piop:zerocheck} on $\mathsf{t}_i \mult (\mathsf{d}_i^{\mathsf{next}} - \mathsf{d}_i)$;
      \item (\emph{broken ties}) the $\mathsf{Pos}$ \cref{piop:sign-check} with width $\alpha$ on the column with activator $\mathsf{t}_{i-1} \mult (1 - \mathsf{t}_i)$ and data $\mathsf{d}_i^{\mathsf{next}} - \mathsf{d}_i$.
    \end{enumerate}
    \item If $\mathsf{strict}$, $\PIOPProver$ and $\PIOPVerifier$ invoke \cref{piop:sumcheck} for $\sum_{x \in \BoolHCube{\mu}} \MLE{\mathsf{t}_m}(x) = 0$; since $\mathsf{t}_m$ is boolean, this forces $\mathsf{t}_m \equiv 0$.
    \item $\PIOPProver$ and $\PIOPVerifier$ output $\Oracle{\MLE{\mathsf{t}_m}}$.
  \end{enumerate}
\end{DefinePIOP}
